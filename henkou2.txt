    try{
        const result=await loadSequentialTiles({
            onStatus:text=>{modelStatus.textContent=text;loadingText.textContent=text;},

                modelGroup.updateMatrixWorld(true);
                renderer.render(scene,camera);
                modelStatus.textContent=`${loadedCount}個表示完了（タイル${batchStartNumber}～${batchEndNumber}）`;
                loadingText.textContent=`${loadedCount}個のモデルを表示しました`;
                progressBar.style.width="100%";

        });
        fitCameraToObject(modelGroup);
        document.getElementById("measurementResult").innerHTML=`<b>読み込み完了</b><br><br>${result.tileCount}個のGLBを読み込みました。`;

    }

import * as THREE from "three";
import { TransformControls } from "three/addons/controls/TransformControls.js";

export class MeasurementRange {
    constructor({ scene, camera, renderer, controls, modelGroup }) {
        this.scene = scene;
        this.camera = camera;
        this.renderer = renderer;
        this.controls = controls;
        this.modelGroup = modelGroup;
        this.type = null;
        this.object = null;
        this.transform = new TransformControls(camera, renderer.domElement);
        this.transform.setSpace("local");
        this.transform.setMode("translate");
        this.scene.add(this.transform.getHelper());
        this.transform.getHelper().visible = false;
        this.transform.addEventListener("dragging-changed", e => {
            this.controls.enabled = !e.value;
        });
    }

    create(type = "box") {
        this.removeObject();
        this.type = type;
        const box = new THREE.Box3().setFromObject(this.modelGroup);
        const center = box.getCenter(new THREE.Vector3());
        const size = box.getSize(new THREE.Vector3());
        const base = Math.max(Math.min(size.x || 1, size.y || 1, size.z || 1), 0.1);

        let geometry;
        if (type === "cylinder") {
            geometry = new THREE.CylinderGeometry(0.5, 0.5, 1, 48, 1, true);
        } else {
            geometry = new THREE.BoxGeometry(1, 1, 1);
        }

        const material = new THREE.MeshBasicMaterial({
            color: 0x00d8ff,
            transparent: true,
            opacity: 0.16,
            depthTest: false,
            side: THREE.DoubleSide
        });
        this.object = new THREE.Mesh(geometry, material);
        this.object.position.copy(center);
        this.object.scale.set(base, base, base);
        this.object.renderOrder = 2000;
        this.scene.add(this.object);

        const edges = new THREE.LineSegments(
            new THREE.EdgesGeometry(geometry),
            new THREE.LineBasicMaterial({ color: 0x00d8ff, depthTest: false, transparent: true, opacity: 0.95 })
        );
        edges.renderOrder = 2001;
        this.object.add(edges);

        this.transform.attach(this.object);
        this.transform.getHelper().visible = true;
        return this.object;
    }

    setMode(mode) {
        if (!this.object) return;
        if (!["translate", "rotate", "scale"].includes(mode)) return;
        this.transform.setMode(mode);
    }

    getRange() {
        if (!this.object || !this.type) return null;
        this.object.updateMatrixWorld(true);
        return {
            type: this.type,
            matrixWorld: this.object.matrixWorld.clone(),
            inverseMatrixWorld: this.object.matrixWorld.clone().invert()
        };
    }

    removeObject() {
        if (!this.object) return;
        this.transform.detach();
        this.scene.remove(this.object);
        this.object.traverse(o => {
            o.geometry?.dispose?.();
            o.material?.dispose?.();
        });
        this.object = null;
        this.type = null;
        this.transform.getHelper().visible = false;
    }

    clear() { this.removeObject(); }
}

import * as THREE from "three";

const EPS = 1e-10;

export function calculateAreaInRange(modelGroup, range, options = {}) {
    if (!range) return { area: 0, meshCount: 0, triangleCount: 0 };
    const cylinderMaxDepth = options.cylinderMaxDepth ?? 7;
    const cylinderTolerance = options.cylinderTolerance ?? 0.002;
    let area = 0, meshCount = 0, triangleCount = 0;

    modelGroup.updateMatrixWorld(true);
    modelGroup.traverse(mesh => {
        if (!mesh.isMesh || !mesh.visible) return;
        const pos = mesh.geometry?.attributes?.position;
        if (!pos) return;
        const index = mesh.geometry.index;
        let meshArea = 0;
        const add = (i0, i1, i2) => {
            const a = new THREE.Vector3().fromBufferAttribute(pos, i0).applyMatrix4(mesh.matrixWorld);
            const b = new THREE.Vector3().fromBufferAttribute(pos, i1).applyMatrix4(mesh.matrixWorld);
            const c = new THREE.Vector3().fromBufferAttribute(pos, i2).applyMatrix4(mesh.matrixWorld);
            let v;
            if (range.type === "box") v = clippedTriangleAreaBox(a, b, c, range.inverseMatrixWorld);
            else v = clippedTriangleAreaCylinder(a, b, c, range.inverseMatrixWorld, cylinderMaxDepth, cylinderTolerance);
            if (v > EPS) { meshArea += v; triangleCount++; }
        };
        if (index) {
            for (let i = 0; i + 2 < index.count; i += 3) add(index.getX(i), index.getX(i+1), index.getX(i+2));
        } else {
            for (let i = 0; i + 2 < pos.count; i += 3) add(i, i+1, i+2);
        }
        if (meshArea > EPS) { area += meshArea; meshCount++; }
    });
    return { area, meshCount, triangleCount };
}

function clippedTriangleAreaBox(aW, bW, cW, inv) {
    let poly = [aW.clone().applyMatrix4(inv), bW.clone().applyMatrix4(inv), cW.clone().applyMatrix4(inv)];
    const planes = [
        [0,  1, 0.5], [0, -1, 0.5],
        [1,  1, 0.5], [1, -1, 0.5],
        [2,  1, 0.5], [2, -1, 0.5]
    ];
    for (const [axis, sign, limit] of planes) {
        poly = clipPolygon(poly, p => limit - sign * p.getComponent(axis), (p1,p2,d1,d2) => p1.clone().lerp(p2, d1/(d1-d2)));
        if (poly.length < 3) return 0;
    }
    const matrix = inv.clone().invert();
    return polygonWorldArea(poly.map(p => p.applyMatrix4(matrix)));
}

function clipPolygon(poly, distanceFn, intersectFn) {
    if (!poly.length) return [];
    const out = [];
    for (let i=0;i<poly.length;i++) {
        const cur=poly[i], prev=poly[(i+poly.length-1)%poly.length];
        const dc=distanceFn(cur), dp=distanceFn(prev);
        const cin=dc>=-EPS, pin=dp>=-EPS;
        if (cin !== pin) out.push(intersectFn(prev,cur,dp,dc));
        if (cin) out.push(cur.clone());
    }
    return out;
}

function polygonWorldArea(poly) {
    if (poly.length < 3) return 0;
    let area=0;
    for (let i=1;i+1<poly.length;i++) area += triangleArea(poly[0],poly[i],poly[i+1]);
    return area;
}

function triangleArea(a,b,c) {
    return new THREE.Vector3().crossVectors(
        new THREE.Vector3().subVectors(b,a),
        new THREE.Vector3().subVectors(c,a)
    ).length()*0.5;
}

function insideCylinderLocal(p) {
    return Math.abs(p.y) <= 0.5 + EPS && (p.x*p.x + p.z*p.z) <= 0.25 + EPS;
}

function clippedTriangleAreaCylinder(aW,bW,cW,inv,maxDepth,tolerance) {
    const matrix = inv.clone().invert();
    const a=aW.clone().applyMatrix4(inv), b=bW.clone().applyMatrix4(inv), c=cW.clone().applyMatrix4(inv);
    return cylinderRecursive(a,b,c,matrix,0,maxDepth,tolerance);
}

function cylinderRecursive(a,b,c,matrix,depth,maxDepth,tolerance) {
    const ia=insideCylinderLocal(a), ib=insideCylinderLocal(b), ic=insideCylinderLocal(c);
    if (ia && ib && ic) return triangleArea(a.clone().applyMatrix4(matrix),b.clone().applyMatrix4(matrix),c.clone().applyMatrix4(matrix));

    const minY=Math.min(a.y,b.y,c.y), maxY=Math.max(a.y,b.y,c.y);
    if (maxY < -0.5 || minY > 0.5) return 0;
    const minX=Math.min(a.x,b.x,c.x), maxX=Math.max(a.x,b.x,c.x);
    const minZ=Math.min(a.z,b.z,c.z), maxZ=Math.max(a.z,b.z,c.z);
    if (minX > 0.5 || maxX < -0.5 || minZ > 0.5 || maxZ < -0.5) return 0;

    const ab=a.distanceTo(b), bc=b.distanceTo(c), ca=c.distanceTo(a);
    if (depth >= maxDepth || Math.max(ab,bc,ca) <= tolerance) {
        const centroid=a.clone().add(b).add(c).multiplyScalar(1/3);
        if (!insideCylinderLocal(centroid)) return 0;
        return triangleArea(a.clone().applyMatrix4(matrix),b.clone().applyMatrix4(matrix),c.clone().applyMatrix4(matrix));
    }

    const abm=a.clone().add(b).multiplyScalar(0.5);
    const bcm=b.clone().add(c).multiplyScalar(0.5);
    const cam=c.clone().add(a).multiplyScalar(0.5);
    return cylinderRecursive(a,abm,cam,matrix,depth+1,maxDepth,tolerance)
         + cylinderRecursive(abm,b,bcm,matrix,depth+1,maxDepth,tolerance)
         + cylinderRecursive(cam,bcm,c,matrix,depth+1,maxDepth,tolerance)
         + cylinderRecursive(abm,bcm,cam,matrix,depth+1,maxDepth,tolerance);
}
